¶ 四、增加触发离线命令词的逻辑

新增 SDK 的文件（路径arcs_mini/apps/arcs-mini/offline_cammand.c)，增加代码：

#include <stdint.h>

#include "sys_init.h"
#include "voice_msg.h"

#define TAG "offline_cmd"
#include "lisa_log.h"

static void offline_command_msg(void *unused, uint32_t msg_id, void *data, uint32_t len, void *user_data)
{
    (void)unused;
    (void)msg_id;
    (void)user_data;

    if (data == NULL || len < sizeof(voice_msg_cloud_recognized_command_t)) {
        LOGW("invalid offline command msg, data:%p len:%u", data, (unsigned int)len);
        return;
    }

    voice_msg_cloud_recognized_command_t *cmd = (voice_msg_cloud_recognized_command_t *)data;
    LOGW("识别到离线命令词： %s", cmd->command);
}

static int offline_command_init(void)
{
    voice_msg_sub(VOICE_MSG_RECOGNIZED_OFFLINE_COMMAND, offline_command_msg, NULL);
    return 0;
}

SYS_INIT(offline_command_init, SYS_INIT_LEVEL_PRE_APPLICATION, 95);

修改 SDK 的文件（路径arcs_mini/apps/arcs-mini/CMakeLists.txt)，增加代码：offline_cammand.c

功能逻辑在arcs_mini/apps/arcs-mini/offline_cammand.c 文件内实现即可。
